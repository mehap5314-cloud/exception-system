📄 الربط بـ Google Sheet — الحفظ التلقائي

1) افتح الشيت نفسه ← قائمة Extensions ← Apps Script (مهم: عشان السكريبت يبقى مربوط بالشيت — ما تعملوش من script.google.com مستقل)
2) امسح اللي في الكود واعمل لصق للكود ده تحت، وبعدين اضغط 💾 الحفظ
3) Deploy ← New deployment ← Web app — خلي Execute as: Me و Who has access: Anyone ← Deploy
4) انسخ الـ Web app URL — ده اللي يبدأ بـ script.google.com/macros/s/… وينتهي بـ /exec
5) الصق الرابط في الخانة دي تحت واضغط حفظ الرابط ← بعد كده جرب 🔌 اختبار الاتصال

⚠️ عنوان صفحة الكود اللي فيه "/projects/…/edit" مش لينك الويب آب — لو حطيته هيظهرلك "فشل رفع السجل". الكود ده بيستخدم تاباتك الموجودة (الأكسبشن · مشكلة موظف · مشكلة منتج) ولو لقأها أسماء إنجليزي هيستخدمها لو مش موجودة يعملها.

// ⚠️ لازم الكود ده يتحط من داخل الشيت نفسه (Extensions ← Apps Script) عشان يبقى مربوط
function doPost(e) {
  var out = { ok: false };
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) { out.err = "NOT-BOUND"; return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); }
    function findSheet(eng, arb) {
      return ss.getSheetByName(eng) || ss.getSheetByName(arb) || ss.insertSheet(eng);
    }
    var CFG = {
      exc: { eng: "Excibsons", arb: "الأكسبشن", headers: ["Date","Agent","Contact Number","Issue","Excibsons","Description","Amount Refund"],
        map: [ ["date", ["Date","التاريخ","تاريخ الفتح"]],
               ["agent", ["Agent","الموظف","اسم الموظف"]],
               ["contact", ["Contact Number","رقم التواصل","رقم العميل"]],
               ["issue", ["Issue","المشكلة","نوع المشكلة"]],
               ["exctype", ["Excibsons","Exceptions","الأكسبشن"]],
               ["desc", ["Description","الوصف","تفاصيل"]],
               ["amount", ["Amount Refund","مبلغ الاسترداد"]] ] },
      emp: { eng: "Employee Issues", arb: "مشكلة موظف", headers: ["Date","Customer","Customer Number","Mobile Type","Branch","Employee Name","Complaint"],
        map: [ ["entryDate", ["Date","التاريخ"]],
               ["name", ["Customer","اسم العميل","العميل"]],
               ["phone", ["Customer Number","رقم العميل"]],
               ["mobileType", ["Mobile Type","نوع الموبايل"]],
               ["branch", ["Branch","الفرع"]],
               ["agent", ["Employee Name","اسم الموظف","الموظف"]],
               ["complaint", ["Complaint","الشكوي","الشكوى","وصف الشكوي"]] ] },
      prod: { eng: "Product Issues", arb: "مشكلة منتج", headers: ["Customer","Mobile Type","Mobile Number","Product","Purchase Date","Problem","Recorded On"],
        map: [ ["name", ["Customer","العميل","اسم العميل"]],
               ["mobileType", ["Mobile Type","نوع الموبايل"]],
               ["mobileNum", ["Mobile Number","رقم الموبايل","IMEI"]],
               ["product", ["Product","المنتج"]],
               ["purchase", ["Purchase Date","تاريخ الشراء"]],
               ["desc", ["Problem","الوصف","المشكلة"]],
               ["entryDate", ["Recorded On","تاريخ التسجيل"]] ] }
    };
    function writeRows(cfg, rows) {
      if (!rows || !rows.length) return;
      var s = findSheet(cfg.eng, cfg.arb);
      if (s.getLastRow() === 0) s.appendRow(cfg.headers);
      var hdr = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
      var layout = [];
      cfg.map.forEach(function (item) {
        var key = item[0], alts = item[1], col = 0;
        for (var i = 0; i !== hdr.length; i++) {
          var h = String(hdr[i]).trim();
          for (var j = 0; j !== alts.length; j++) {
            if (h && h === alts[j]) { col = i + 1; break; }
          }
          if (col) break;
        }
        if (!col) { col = s.getLastColumn() + 1; s.getRange(1, col).setValue(alts[0]); hdr.push(alts[0]); }
        layout.push([key, col]);
      });
      var maxCol = s.getLastColumn();
      rows.forEach(function (r) {
        var row = [];
        for (var i = 0; i !== maxCol; i++) row.push("");
        layout.forEach(function (p) { row[p[1] - 1] = (r[p[0]] === undefined || r[p[0]] === null) ? "" : r[p[0]]; });
        s.appendRow(row);
      });
    }
    if (data.type === "exc") writeRows(CFG.exc, [data]);
    else if (data.type === "emp") writeRows(CFG.emp, [data]);
    else if (data.type === "prod") writeRows(CFG.prod, [data]);
    else if (data.type === "syncAll") {
      writeRows(CFG.exc, data.exc);
      writeRows(CFG.emp, data.emp);
      writeRows(CFG.prod, data.prod);
    }
    out.ok = true;
  } catch (err) { out.err = String(err); }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var out = { ok: false, err: "", data: { exc: [], emp: [], prod: [] } };
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) { out.err = "NOT-BOUND"; return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); }
    var CFG = {
      exc: { eng: "Excibsons", arb: "الأكسبشن",
        map: [ ["date", ["Date","التاريخ","تاريخ الفتح"]],
               ["agent", ["Agent","الموظف","اسم الموظف"]],
               ["contact", ["Contact Number","رقم التواصل","رقم العميل"]],
               ["issue", ["Issue","المشكلة","نوع المشكلة"]],
               ["exctype", ["Excibsons","Exceptions","الأكسبشن"]],
               ["desc", ["Description","الوصف","تفاصيل"]],
               ["amount", ["Amount Refund","مبلغ الاسترداد"]] ] },
      emp: { eng: "Employee Issues", arb: "مشكلة موظف",
        map: [ ["entryDate", ["Date","التاريخ"]],
               ["name", ["Customer","اسم العميل","العميل"]],
               ["phone", ["Customer Number","رقم العميل"]],
               ["mobileType", ["Mobile Type","نوع الموبايل"]],
               ["branch", ["Branch","الفرع"]],
               ["agent", ["Employee Name","اسم الموظف","الموظف"]],
               ["complaint", ["Complaint","الشكوي","الشكوى","وصف الشكوي"]] ] },
      prod: { eng: "Product Issues", arb: "مشكلة منتج",
        map: [ ["name", ["Customer","العميل","اسم العميل"]],
               ["mobileType", ["Mobile Type","نوع الموبايل"]],
               ["mobileNum", ["Mobile Number","رقم الموبايل","IMEI"]],
               ["product", ["Product","المنتج"]],
               ["purchase", ["Purchase Date","تاريخ الشراء"]],
               ["desc", ["Problem","الوصف","المشكلة"]],
               ["entryDate", ["Recorded On","تاريخ التسجيل"]] ] }
    };
    function readRows(cfg) {
      var s = ss.getSheetByName(cfg.eng) || ss.getSheetByName(cfg.arb);
      if (!s || s.getLastRow() < 2) return [];
      var cols = s.getLastColumn();
      var headers = s.getRange(1, 1, 1, cols).getValues()[0];
      var layout = [];
      cfg.map.forEach(function (item) {
        var alts = item[1], col = 0;
        for (var i = 0; i !== headers.length; i++) {
          var h = String(headers[i]).trim();
          for (var j = 0; j !== alts.length; j++) { if (h === alts[j]) { col = i + 1; break; } }
          if (col) break;
        }
        if (col) layout.push([item[0], col]);
      });
      var values = s.getRange(2, 1, s.getLastRow() - 1, cols).getValues();
      return values.map(function (row) {
        var rec = {};
        layout.forEach(function (p) { rec[p[0]] = row[p[1] - 1]; });
        if (rec.amount !== undefined && rec.amount !== "") rec.amount = Number(rec.amount) || 0;
        return rec;
      }).filter(function (r) { return Object.keys(r).some(function (k) { return String(r[k] || "").trim() !== ""; }); });
    }
    out.data.exc = readRows(CFG.exc);
    out.data.emp = readRows(CFG.emp);
    out.data.prod = readRows(CFG.prod);
    out.ok = true;
  } catch (err) { out.err = String(err); }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

• أول ما تحفظ أكسبشن أو شكوي أو مشكلة والرابط متفعل، هيتحفظ صف تلقائي في الشيت (النقطة الخضراء ☁ في جداول السجلات معناها متزامن).
• أي سجل فشل رفعه (نت مقطوع مثلًا) هيتوسم بالبرتقالي ☁ وتقدر تضغط "مزامنة الكل الآن" بدون تكرار.
• عند تعديل سجل هيتضاف صف جديد في الشيت (الشيت بيضيف فقط) — الحذف بيتم يدويًا من الشيت نفسه.
